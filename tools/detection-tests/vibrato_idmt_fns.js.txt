let yinBuf=null;
function yin(buf,sr){
  const half=buf.length>>1;const maxTau=Math.min(half-1,Math.floor(sr/70));const minTau=Math.max(2,Math.floor(sr/1300));
  if(!yinBuf||yinBuf.length<maxTau+2)yinBuf=new Float32Array(maxTau+2);
  const d=yinBuf;d[0]=1;let run=0;
  for(let tau=1;tau<=maxTau;tau++){let sum=0;for(let i=0;i<half;i++){const x=buf[i]-buf[i+tau];sum+=x*x;}run+=sum;d[tau]=run>0?sum*tau/run:1;}
  let tau=-1;
  for(let t=minTau;t<=maxTau;t++){if(d[t]<0.15){while(t+1<=maxTau&&d[t+1]<d[t])t++;tau=t;break;}}
  if(tau<0){let mi=minTau;for(let t=minTau;t<=maxTau;t++)if(d[t]<d[mi])mi=t;if(d[mi]<0.3)tau=mi;else return -1;}
  let b=tau;if(tau>1&&tau<maxTau){const x0=d[tau-1],x1=d[tau],x2=d[tau+1];const den=x0-2*x1+x2;if(Math.abs(den)>1e-9)b=tau+0.5*(x0-x2)/den;}
  return sr/b;
}
function vibMeasure(t0,t1){const A=G.vibC;if(!A||t1-t0<550)return null;
  const xs=[],ys=[];for(let k=0;k<A.length;k+=2)if(A[k]>=t0&&A[k]<=t1){xs.push(A[k]);ys.push(A[k+1]);}
  if(xs.length<(t1-t0)/16.7*0.6)return null;
  const so=ys.slice().sort((a,b)=>a-b),med=so[so.length>>1];const kx=[],ky=[];for(let k=0;k<xs.length;k++)if(Math.abs(ys[k]-med)<=150){kx.push(xs[k]);ky.push(ys[k]);} /* the next note starting, or an octave slip, is not vibrato */
  if(kx.length<(t1-t0)/16.7*0.55)return null;
  const N=Math.floor((t1-t0)/20),g=new Float64Array(N);let j=0;
  for(let n=0;n<N;n++){const t=t0+n*20;while(j<kx.length-2&&kx[j+1]<t)j++;const x0=kx[j],x1=kx[j+1],f=x1>x0?Math.max(0,Math.min(1,(t-x0)/(x1-x0))):0;g[n]=ky[j]+(ky[j+1]-ky[j])*f;}
  let sx=0,sy=0,sxx=0,sxy=0;for(let n=0;n<N;n++){sx+=n;sy+=g[n];sxx+=n*n;sxy+=n*g[n];}const bsl=(N*sxy-sx*sy)/(N*sxx-sx*sx),b0=(sy-bsl*sx)/N;
  const w=new Float64Array(N);let sw=0;for(let n=0;n<N;n++){w[n]=0.5-0.5*Math.cos(2*Math.PI*(n+0.5)/N);sw+=w[n];g[n]=(g[n]-b0-bsl*n)*w[n];}
  let best=0,bf=0;for(let f=3.5;f<=9.01;f+=0.25){let re=0,im=0;for(let n=0;n<N;n++){const ph=2*Math.PI*f*n*0.02;re+=g[n]*Math.cos(ph);im-=g[n]*Math.sin(ph);}const a=2*Math.hypot(re,im)/sw;if(a>best){best=a;bf=f;}}
  return {d:2*best,f:bf};}
